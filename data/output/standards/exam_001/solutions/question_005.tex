% Standard solution for exam_001 / question_005 (continuity at a point)
% f continuous at x=1 for the continuous extension of (x^2-1)/(x-1)
% lim_{x \to 1} f(x) = f(1) = 2
\begin{aligned}
\lim_{x \to 1} \frac{x^2-1}{x-1}
&= \lim_{x \to 1} (x+1) \\
&= 2 \\
f(1) &= 2
\end{aligned}

% final answer
f continuous at 1 with f(1)=2
