% Standard solution for exam_001 / question_009 (matrix ops: add + transpose)
% A + B and A^T for 2x2 numeric matrices
\begin{aligned}
\begin{pmatrix}1 & 2 \\ 3 & 4\end{pmatrix}
+
\begin{pmatrix}5 & 6 \\ 7 & 8\end{pmatrix}
&=
\begin{pmatrix}6 & 8 \\ 10 & 12\end{pmatrix}
\\
\begin{pmatrix}1 & 2 \\ 3 & 4\end{pmatrix}^{T}
&=
\begin{pmatrix}1 & 3 \\ 2 & 4\end{pmatrix}
\end{aligned}

% final answer
\begin{pmatrix}6 & 8 \\ 10 & 12\end{pmatrix}
